    <script>

        /**

         * 3D SCENE CONFIGURATION

         */

        let scene, camera, renderer, dome, wireframe, particles;

        let clock = new THREE.Clock();

        let isPlaying = false;

        let targetIntensity = 0;

        let currentIntensity = 0;


        function init() {

            scene = new THREE.Scene();

            camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);

            

            renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });

            renderer.setSize(window.innerWidth, window.innerHeight);

            renderer.setPixelRatio(window.devicePixelRatio);

            document.getElementById('canvas-container').appendChild(renderer.domElement);


            // Lighting

            const ambientLight = new THREE.AmbientLight(0xffffff, 0.8);

            scene.add(ambientLight);


            const pointLight = new THREE.PointLight(0x0000ff, 1, 100);

            pointLight.position.set(10, 10, 10);

            scene.add(pointLight);


            // Geodesic Dome Geometry

            const geometry = new THREE.IcosahedronGeometry(4, 3);

            const material = new THREE.MeshPhongMaterial({

                color: 0xffffff,

                transparent: true,

                opacity: 0.1,

                shininess: 100

            });

            

            dome = new THREE.Mesh(geometry, material);

            scene.add(dome);


            // Structure Outline (Wireframe)

            const wireframeGeo = new THREE.WireframeGeometry(geometry);

            const wireframeMat = new THREE.LineBasicMaterial({ 

                color: 0x000000, 

                transparent: true, 

                opacity: 0.2 

            });

            wireframe = new THREE.LineSegments(wireframeGeo, wireframeMat);

            dome.add(wireframe);


            // Ambient Background Particles

            const partGeo = new THREE.BufferGeometry();

            const partCount = 500;

            const posArray = new Float32Array(partCount * 3);

            

            for(let i=0; i < partCount * 3; i++) {

                posArray[i] = (Math.random() - 0.5) * 20;

            }

            partGeo.setAttribute('position', new THREE.BufferAttribute(posArray, 3));

            

            const partMat = new THREE.PointsMaterial({

                size: 0.02,

                color: 0x000000,

                transparent: true,

                opacity: 0.4

            });

            

            particles = new THREE.Points(partGeo, partMat);

            scene.add(particles);


            camera.position.z = 10;

            camera.position.y = 1;


            window.addEventListener('resize', onWindowResize, false);

            

            // Interaction Controller

            document.getElementById('toggle-audio').addEventListener('click', function() {

                isPlaying = !isPlaying;

                this.innerText = isPlaying ? "PAUSE SESSION" : "START SESSION";

                this.classList.toggle('bg-blue-600');

            });


            animate();

        }


        function onWindowResize() {

            camera.aspect = window.innerWidth / window.innerHeight;

            camera.updateProjectionMatrix();

            renderer.setSize(window.innerWidth, window.innerHeight);

        }


        /**

         * ANIMATION LOOP

         * Handles vertex displacement and rotation

         */

        function animate() {

            requestAnimationFrame(animate);

            const t = clock.getElapsedTime();


            // Interpolate intensity for smooth transitions

            if (isPlaying) {

                targetIntensity = 0.5 + Math.sin(t * 2) * 0.4;

            } else {

                targetIntensity = 0;

            }

            currentIntensity += (targetIntensity - currentIntensity) * 0.1;


            // Global Rotation

            dome.rotation.y += 0.002;

            dome.rotation.x = Math.sin(t * 0.5) * 0.1;


            // Audio-Reactive Vertex Animation

            const positionAttribute = dome.geometry.getAttribute('position');

            

            for (let i = 0; i < positionAttribute.count; i++) {

                const x = positionAttribute.getX(i);

                const y = positionAttribute.getY(i);

                const z = positionAttribute.getZ(i);

                

                const vector = new THREE.Vector3(x, y, z);

                vector.normalize();

                

                // Simulating frequency oscillation

                const noise = Math.sin(x * 2 + t * 4) * Math.cos(y * 2 + t * 3) * currentIntensity;

                const newScale = 4 + noise;

                

                vector.multiplyScalar(newScale);

                positionAttribute.setXYZ(i, vector.x, vector.y, vector.z);

            }

            positionAttribute.needsUpdate = true;

            

            // Update wireframe to match deformed geometry

            wireframe.geometry.dispose();

            wireframe.geometry = new THREE.WireframeGeometry(dome.geometry);


            // Particle drift

            particles.rotation.y -= 0.001;

            particles.position.y = Math.sin(t) * 0.2;


            // Smooth parallax camera

            camera.position.x += (Math.sin(t * 0.3) * 0.5 - camera.position.x) * 0.05;


            renderer.render(scene, camera);

        }


        window.onload = init;

    </script>
